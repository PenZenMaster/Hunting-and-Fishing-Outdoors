vendor\bin\phpcs --standard=phpcs.xml.dist && vendor\bin\phpunit

### Pre-Commit Checklist

1. **Code Standards**: `composer phpcs` (must pass)
2. **Unit Tests**: `composer test` (must pass)
3. **Security Review**: Verify sanitization, escaping, nonces, capabilities
4. **I18n Check**: All user-facing strings wrapped with text domain `hnfo`
5. **No Debug Code**: Remove `var_dump()`, `print_r()`, `console.log()`, etc.

## WordPress Coding Standards (Mandatory)

### File Structure

```php
<?php
/**
 * Module/Script Name: [Name]
 * Path: [Full path with filename]
 *
 * Description:
 * [Brief summary]
 *
 * @package    HNFO
 * @author     Rank Rocket Co
 * @copyright  2026 Rank Rocket Co - All Rights Reserved
 * @version    1.00
 *
 * Created Date: 2026-02-02
 * Last Modified Date: [Auto-filled]
 *
 * Comments:
 * - Initial version v1.00
 */

// Exit if accessed directly
if ( ! defined( 'ABSPATH' ) ) {
    exit;
}
```

### Naming Conventions

- **Functions**: `hnfo_function_name()` (prefix required)
- **Classes**: `HNFO_Class_Name` (prefix required)
- **Constants**: `HNFO_CONSTANT_NAME` (prefix required)
- **Variables**: `$lowercase_with_underscores`
- **Files**: `class-my-class.php`, `my-functions.php`

### Security Rules (Non-Negotiable)

#### Input Sanitization

```php
// Text input
$text = sanitize_text_field( $_POST['field'] );

// Email
$email = sanitize_email( $_POST['email'] );

// URL
$url = sanitize_url( $_POST['url'] );

// Textarea
$content = sanitize_textarea_field( $_POST['content'] );

// Array
$array = array_map( 'sanitize_text_field', $_POST['array'] );
```

#### Output Escaping

```php
// HTML content
echo esc_html( $text );

// Attributes
echo '<div class="' . esc_attr( $class ) . '">';

// URLs
echo '<a href="' . esc_url( $url ) . '">';

// JavaScript
echo '<script>var data = ' . wp_json_encode( $data ) . ';</script>';

// Translation with escaping
echo esc_html__( 'Text', 'hnfo' );
echo esc_html_e( 'Text', 'hnfo' );
```

#### Database Queries

```php
// CORRECT - Use $wpdb->prepare()
global $wpdb;
$results = $wpdb->get_results(
    $wpdb->prepare(
        "SELECT * FROM {$wpdb->prefix}table WHERE id = %d AND name = %s",
        $id,
        $name
    )
);

// NEVER - Direct query with user input
$results = $wpdb->get_results( "SELECT * FROM table WHERE id = {$id}" ); // WRONG!
```

#### Nonces

```php
// Form output
wp_nonce_field( 'hnfo_action', 'hnfo_nonce' );

// Verification
if ( ! isset( $_POST['hnfo_nonce'] ) || ! wp_verify_nonce( $_POST['hnfo_nonce'], 'hnfo_action' ) ) {
    wp_die( esc_html__( 'Security check failed', 'hnfo' ) );
}
```

#### Capability Checks

```php
// Check user permissions
if ( ! current_user_can( 'manage_options' ) ) {
    wp_die( esc_html__( 'You do not have permission', 'hnfo' ) );
}
```

### Internationalization (Required)

```php
// Simple translation
__( 'Text to translate', 'hnfo' );

// Translation with echo
_e( 'Text to translate', 'hnfo' );

// Plural forms
_n( 'One item', '%d items', $count, 'hnfo' );

// Context-specific
_x( 'Post', 'noun', 'hnfo' );

// With escaping (preferred)
esc_html__( 'Text', 'hnfo' );
esc_html_e( 'Text', 'hnfo' );
esc_attr__( 'Text', 'hnfo' );
```

**Text Domain**: Always use `hnfo` for this project.

### WordPress Hooks

```php
// Actions
add_action( 'init', 'hnfo_initialize_plugin' );
add_action( 'admin_menu', 'hnfo_add_admin_menu' );
add_action( 'wp_enqueue_scripts', 'hnfo_enqueue_assets' );

// Filters
add_filter( 'the_content', 'hnfo_modify_content' );
add_filter( 'query_vars', 'hnfo_add_query_vars' );

// Priority and args
add_action( 'init', 'hnfo_function', 10, 2 ); // Priority 10, 2 arguments
```

### Enqueuing Assets

```php
/**
 * Enqueue scripts and styles.
 */
function hnfo_enqueue_assets() {
    // CSS
    wp_enqueue_style(
        'hnfo-style',
        plugin_dir_url( __FILE__ ) . 'assets/css/style.css',
        array(),
        '1.0.0'
    );

    // JavaScript
    wp_enqueue_script(
        'hnfo-script',
        plugin_dir_url( __FILE__ ) . 'assets/js/script.js',
        array( 'jquery' ),
        '1.0.0',
        true // In footer
    );

    // Localize script
    wp_localize_script(
        'hnfo-script',
        'hnfoData',
        array(
            'ajaxUrl' => admin_url( 'admin-ajax.php' ),
            'nonce'   => wp_create_nonce( 'hnfo_ajax' ),
        )
    );
}
add_action( 'wp_enqueue_scripts', 'hnfo_enqueue_assets' );
```

### AJAX Handlers

```php
/**
 * Handle AJAX request.
 */
function hnfo_ajax_handler() {
    // Verify nonce
    check_ajax_referer( 'hnfo_ajax', 'nonce' );

    // Check capabilities
    if ( ! current_user_can( 'edit_posts' ) ) {
        wp_send_json_error( __( 'Permission denied', 'hnfo' ) );
    }

    // Sanitize input
    $data = sanitize_text_field( $_POST['data'] );

    // Process...

    // Send response
    wp_send_json_success( array( 'message' => __( 'Success', 'hnfo' ) ) );
}
add_action( 'wp_ajax_hnfo_action', 'hnfo_ajax_handler' );
add_action( 'wp_ajax_nopriv_hnfo_action', 'hnfo_ajax_handler' ); // For logged-out users
```

### Custom Post Types

```php
/**
 * Register custom post type.
 */
function hnfo_register_post_type() {
    register_post_type(
        'hnfo_item',
        array(
            'labels'       => array(
                'name'          => __( 'Items', 'hnfo' ),
                'singular_name' => __( 'Item', 'hnfo' ),
            ),
            'public'       => true,
            'has_archive'  => true,
            'rewrite'      => array( 'slug' => 'items' ),
            'supports'     => array( 'title', 'editor', 'thumbnail' ),
            'show_in_rest' => true, // For Gutenberg
        )
    );
}
add_action( 'init', 'hnfo_register_post_type' );
```

## Common WordPress Patterns

### Plugin Activation/Deactivation

```php
/**
 * Plugin activation.
 */
function hnfo_activate() {
    // Create tables, set options, etc.
    flush_rewrite_rules();
}
register_activation_hook( __FILE__, 'hnfo_activate' );

/**
 * Plugin deactivation.
 */
function hnfo_deactivate() {
    flush_rewrite_rules();
}
register_deactivation_hook( __FILE__, 'hnfo_deactivate' );
```

### Options API

```php
// Get option
$value = get_option( 'hnfo_option', 'default_value' );

// Update option
update_option( 'hnfo_option', $value );

// Delete option
delete_option( 'hnfo_option' );
```

### Meta Data

```php
// Post meta
add_post_meta( $post_id, 'hnfo_key', $value, true );
$value = get_post_meta( $post_id, 'hnfo_key', true );
update_post_meta( $post_id, 'hnfo_key', $value );
delete_post_meta( $post_id, 'hnfo_key' );

// User meta
add_user_meta( $user_id, 'hnfo_key', $value, true );
$value = get_user_meta( $user_id, 'hnfo_key', true );
```

## Forbidden Patterns

### NEVER Use These

```php
// Direct database access
mysql_query( ... ); // Use $wpdb instead

// Unsanitized output
echo $_POST['data']; // Always sanitize/escape

// Direct file inclusion
include( $_GET['file'] ); // Security risk

// Direct SQL
$wpdb->query( "DELETE FROM table WHERE id = {$id}" ); // Use prepare()

// Eval
eval( $code ); // Never use eval

// Generic function names
function my_function() {} // Use hnfo_ prefix

// Inline styles/scripts (without reason)
<script>...</script> // Use wp_enqueue_script()

// Direct constant checks without ABSPATH
// Missing: if ( ! defined( 'ABSPATH' ) ) exit;
```

## Development Workflow

### 1. Before Writing Code

- Read existing code patterns
- Check `WORDPRESS-STANDARDS.md`
- Verify text domain and prefixes

### 2. While Writing Code

- Use proper prefixes (`hnfo_`)
- Sanitize ALL inputs
- Escape ALL outputs
- Add nonces to forms
- Check capabilities
- Wrap strings for translation
- Add inline documentation

### 3. Before Committing

```bash
# Run quality gate
vendor\bin\phpcs --standard=phpcs.xml.dist

# Auto-fix what can be fixed
vendor\bin\phpcbf --standard=phpcs.xml.dist

# Run tests
vendor\bin\phpunit

# If all pass, commit
git add -A
git commit -m "feat: descriptive message"
```
